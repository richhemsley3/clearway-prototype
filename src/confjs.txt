 /* ── Northgate's configuration ─────────────────────────────────── */
 var cfgSec='all';
 function drawConfig(){
   var internal = ws==='cw';
   var locked=0, open=0, unset=0;
   CONFIG.forEach(function(s){ s.rows.forEach(function(r){ if(r[4]==='locked') locked++; else open++; if(r[1]==='Not set') unset++; }) });
   $('#cfSub').textContent=(internal?'Northgate Financial · ':'')+'set on 14 August · '+locked+' items are now permanent · '+open+' can still change · '+unset+' not set';
   var K=[['all','Everything',locked+open]].concat(CONFIG.map(function(s){ return [s.s, s.s, s.rows.length] }));
   $('#cfSeg').innerHTML='<span class="seg">'+K.map(function(t){ return '<button data-c="'+t[0]+'" aria-pressed="'+(cfgSec===t[0])+'">'+t[1]+'<span class="ct">'+t[2]+'</span></button>' }).join('')
    +'</span><div class="rt"><span class="lnk" id="cfExp">Export as JSON</span></div>';
   $$('#cfSeg button').forEach(function(b){ b.addEventListener('click',function(){ anchored($('#cfSeg'), function(){ cfgSec=b.dataset.c; drawConfig(); }) }) });
   $('#cfExp').addEventListener('click',function(){ toast('northgate-471288-config.json — the same file the sandbox reads.') });

   var secs=CONFIG.filter(function(s){ return cfgSec==='all'||s.s===cfgSec });
   $('#cfBody').innerHTML=secs.map(function(s){
     return '<div class="sech"'+(s===secs[0]?' style="margin-top:26px"':'')+'><h2>'+esc(s.s)+'</h2><span class="t13">'+s.rows.filter(function(r){return r[4]==='locked'}).length+' permanent</span></div>'
      +'<table class="tb" style="margin-top:10px"><thead><tr><th style="width:38%">Setting</th><th>Value</th><th class="opt" style="width:84px">Set by</th><th class="opt2" style="width:76px; white-space:nowrap">When</th><th class="n" style="width:110px"></th></tr></thead><tbody>'
      +s.rows.map(function(r,i){
        var unset=r[1]==='Not set', mono=/^https|^\d{3}\.|^CN=|•••• \d{4}|^\d{6} ·/.test(r[1]);
        return '<tr data-s="'+esc(s.s)+'" data-i="'+i+'"'+(unset?' class="hot"':'')+'><td>'+esc(r[0])+(r[5]?'<span class="s2" style="max-width:52ch">'+esc(r[5])+'</span>':'')+'</td>'
         +'<td>'+(mono?'<span class="key">'+esc(r[1])+'</span>':unset?'<span class="st wa">Not set</span>':esc(r[1]))+'</td>'
         +'<td class="opt">'+turn(r[2]==='GPN'?'cw':'you')+'</td><td class="opt2 mute" style="white-space:nowrap">'+esc(r[3])+'</td>'
         +'<td class="n">'+(r[4]==='locked'?'<span class="t13" style="color:var(--g550)">Permanent</span>':'<span class="lnk">'+(unset?'Set it':'Change')+'</span>')+'</td></tr>' }).join('')
      +'</tbody></table>';
   }).join('')
   +'<div class="txtwo" style="margin-top:44px">'
   +'<div class="note plain"><div class="k">What permanent means</div><p>Product type, BIN, protocol and currency are written into the range assignment at certification. Changing any of them after 19 September is a new BIN, a new certification and, for plastic, new cards in the post. Everything else changes with a message to your integration manager and takes effect on the next cycle.</p></div>'
   +'<div class="note'+(internal?'':' plain')+'"><div class="k">'+(internal?'One thing unset, one thing expiring':'Before go live')+'</div><p>No failover endpoint is configured. It is optional before activation, but the first settlement night without one is the night you find out why it is recommended. And the mutual TLS certificate expires on 21 September.</p><div class="act"><span class="lnk" id="cfFail">Set the failover endpoint</span></div></div></div>';
   $$('#cfBody tr[data-s]').forEach(function(tr){ tr.addEventListener('click',function(){
     var s=CONFIG.filter(function(x){return x.s===tr.dataset.s})[0], r=s.rows[+tr.dataset.i];
     panel(r[0], s.s+' · set by '+(r[2]==='GPN'?'Global Payment Network':'Northgate')+(r[3]!=='—'?' on '+r[3]:''),
       '<div class="rec" style="margin-bottom:20px"><div class="rr"><span class="l">Value</span><span class="v" style="max-width:34ch">'+esc(r[1])+'</span></div>'
       +'<div class="rr"><span class="l">Can change</span><span class="v">'+(r[4]==='locked'?'No — permanent at certification':'Yes, effective next cycle')+'</span></div></div>'
       +(r[5]?'<p style="font-size:15px; color:var(--g650); line-height:1.6">'+esc(r[5])+'</p>':'')
       +(r[4]==='open'?'<div class="field" style="margin-top:22px"><label>New value</label><input class="in" value="'+esc(r[1]==='Not set'?'':r[1])+'"></div>':''),
       r[4]==='locked'?null:(internal?'Change it for Northgate':'Request the change'),
       function(){ toast(internal?'Changed. Takes effect on the next cycle, and Northgate is told.':'Sent to Priya Raman. Changes take effect on the next cycle.') });
   }) });
   $('#cfFail').addEventListener('click',function(){ toast('Form opened. It has to answer a signed ping before we save it.') });
 }
