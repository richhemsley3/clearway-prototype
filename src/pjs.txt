 function contractingClosed(){
   var a=QUEUE.filter(function(q){return q.k==='agr'})[0], r=QUEUE.filter(function(q){return q.k==='risk'})[0];
   return a.dstate==='done' && r.dstate==='done';
 }

 function passCase(id){
   var f=FAILS.filter(function(x){return x.id===id})[0]; if(!f) return;
   var s=SUITE.filter(function(x){return x.k===f.suite})[0];
   if(s){ s.pass++; s.fail--; }
   f.passed=true; f.act=f.fix; f.run='Just now';
   FAILS=FAILS.filter(function(x){ return x.id!==id }).concat([f]);
   refreshNav();
 }
 function openFails(){ return FAILS.filter(function(f){ return !f.passed }) }
 function casePassed(id){ var f=FAILS.filter(function(x){return x.id===id})[0]; return !f || !!f.passed }

 function subFor(q){
   if(q.state==='partial') return ' · one of two signatures in';
   if(q.state==='rejected') return ' · sent back, address check';
   if(q.state==='review') return ' · with our reviewer';
   if(q.k==='ack') return ' · '+(6-polTodo(acc))+' of six';
   return '';
 }
 function actionFor(q){
   if(q.state==='rejected') return 'Send a replacement';
   if(q.who==='cw') return 'Ask Dara where it is';
   if(q.kind==='Signature') return 'Open the agreement';
   if(q.kind==='Acknowledgement') return 'Read the outstanding four';
   return 'Send the document';
 }
 function shortAction(q){
   if(q.dstate==='done') return 'View';
   if(q.who==='cw') return 'Chase';
   if(q.state==='rejected') return 'Replace';
   if(q.kind==='Signature') return 'Sign';
   if(q.kind==='Acknowledgement') return 'Read';
   return 'Send';
 }

 function rerun(id){

       passCase(id);

     if(curView==='dv-tool') drawTool(); else if(curView==='pd-integr') drawIntegr();
   }, 1100);
 }
