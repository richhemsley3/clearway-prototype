 function ruleBlocks(items){
   /* items: [{p, st, sub, act, actCls, onAct, onOpen}] */
   return '<div class="docs">'+items.map(function(it,i){ var p=it.p;
     return '<div class="doc'+(it.hot?' hot':'')+'" data-i="'+i+'"><div class="dot"></div>'
      +'<div class="body"><h3>'+esc(p[0])+'</h3><div class="meta">'+esc(p[1])+' · effective '+esc(p[2])+' · '+p[3]+' pages</div>'
      +'<p>'+esc(DESC[p[0]]||'')+'</p>'
      +(it.change?'<div class="chg"><span class="k">What changed · '+esc(it.change[0])+'</span>'+esc(it.change[1])+'</div>':'')
      +'</div>'
      +'<div class="side"><span class="st '+it.stc+'">'+esc(it.stl)+'</span><span class="sub">'+esc(it.sub||'')+'</span>'
      +(it.act?'<button class="btn'+(it.actCls?' '+it.actCls:'')+' sm" data-act="'+i+'">'+esc(it.act)+'</button>':'')
      +'<span class="lnk" data-open="'+i+'">Open the document</span></div></div>' }).join('')+'</div>';
 }

 drawHub=function(){

   var changed=all.filter(function(d){ return !d.guide && d.what.indexOf('No change')!==0 });
   $('#hubSeg').innerHTML='';
   var groups=[['Changed this month',changed],['Policies',all.filter(function(d){return d.kind==='Policy'})],['Specification and reference',all.filter(function(d){return d.kind==='Specification'||d.kind==='Reference'})],['Guides',all.filter(function(d){return d.kind==='Guide'})],['Release notes',all.filter(function(d){return d.kind==='Notes'})]];
