 /* ── small marks ──────────────────────────────────────────────── */
 function spark(v,w,h,opts){
   opts=opts||{}; w=w||120; h=h||28;
   var mn=Math.min.apply(null,v), mx=Math.max.apply(null,v), rg=(mx-mn)||1;
   var x=function(i){ return (i/(v.length-1))*(w-4)+2 }, y=function(n){ return 2+(1-(n-mn)/rg)*(h-4) };
   var d=v.map(function(n,i){ return (i?'L':'M')+x(i).toFixed(1)+' '+y(n).toFixed(1) }).join(' ');
   var col=opts.color||getComputedStyle(document.documentElement).getPropertyValue('--navy').trim();
   return '<svg class="spark" viewBox="0 0 '+w+' '+h+'" width="'+w+'" height="'+h+'" aria-hidden="true"><path d="'+d+'" fill="none" stroke="'+col+'" stroke-width="1.5" stroke-linejoin="round"/>'
    +'<circle cx="'+x(v.length-1).toFixed(1)+'" cy="'+y(v[v.length-1]).toFixed(1)+'" r="2.5" fill="'+col+'"/></svg>';
 }
 function hbars(v,opts){
   opts=opts||{}; var W=opts.w||380, H=opts.h||150, L=36, top=8, base=H-22, mx=Math.max.apply(null,v), n=v.length, slot=(W-L)/n, bw=slot*0.62, out='';
   var cv=function(k){ return getComputedStyle(document.documentElement).getPropertyValue(k).trim() };
   for(var g=0;g<3;g++){ var val=mx-g*mx/2, yy=top+(1-val/mx)*(base-top);
     out+='<line x1="'+L+'" x2="'+W+'" y1="'+yy.toFixed(1)+'" y2="'+yy.toFixed(1)+'" stroke="'+cv('--g150')+'" stroke-dasharray="2 4"/><text x="0" y="'+(yy+4).toFixed(1)+'" font-size="11" fill="'+cv('--g400')+'">'+(opts.fmt?opts.fmt(val):Math.round(val))+'</text>'; }
   v.forEach(function(val,i){ var h=val/mx*(base-top), x=L+i*slot+(slot-bw)/2, hot=opts.hot&&opts.hot(i);
     out+='<rect x="'+x.toFixed(1)+'" y="'+(base-h).toFixed(1)+'" width="'+bw.toFixed(1)+'" height="'+h.toFixed(1)+'" rx="1" fill="'+(hot?cv('--warn'):cv('--navy'))+'" opacity="'+(hot?1:(i>opts.nowIdx?0.2:0.9))+'"/>'; });
   [0,6,12,18,23].forEach(function(i){ out+='<text x="'+(L+i*slot+slot/2).toFixed(1)+'" y="'+(H-6)+'" text-anchor="middle" font-size="11" fill="'+cv('--g400')+'">'+(i<10?'0'+i:i)+':00</text>' });
   return '<svg viewBox="0 0 '+W+' '+H+'" style="width:100%; height:auto; display:block; font-family:var(--ui)" aria-hidden="true">'+out+'</svg>';
 }

 /* ── MESSAGES · two panes ─────────────────────────────────────── */
 /* rows: [{id,t,about,w,st,last,msgs,a}] · o: {who, status(t)→[cls,label], open, onOpen, reply(t,text), resolve(t), host} */
 function threadPane(o){
   var rank=function(t){ var s=o.status(t)[0]; return s==='wa'?0:s==='nt'?1:2 };
   var rows=o.rows.slice().sort(function(a,b){ return rank(a)-rank(b) }), cur=rows.filter(function(t){return t.id===o.open})[0]||rows[0];
   if(cur && cur.id!==o.open) o.onOpen(cur.id);
   var list=rows.map(function(t){ var s=o.status(t);
     return '<a data-t="'+t.id+'" class="'+(cur&&t.id===cur.id?'on':'')+'"><span class="t">'+esc(t.t)+'</span>'
      +'<span class="m"><span>'+(o.who2?'<span class="mute">'+esc(o.who2(t))+' · </span>':'')+'<span class="st '+s[0]+'">'+esc(s[1])+'</span></span><span class="mute nw">'+esc(t.last)+'</span></span></a>' }).join('');
   var thread='';
   if(cur){
     var s=o.status(cur);
     thread='<div class="th"><div class="hd"><h2>'+esc(cur.t)+'</h2><div class="meta"><span>'+esc(cur.about||'')+'</span><span>·</span><span>with '+esc(cur.w)+'</span><span>·</span><span class="st '+s[0]+'">'+esc(s[1])+'</span></div></div>'
      +cur.msgs.slice().reverse().map(function(m){ var us=m[1]==='Global Payment Network';
         return '<div class="msg'+(us?' us':'')+'"><div class="who">'+esc(m[0])+' <span class="mute">· '+esc(m[1])+' · '+esc(m[2])+'</span></div><p>'+esc(m[3])+'</p></div>' }).join('')
      +'<div class="reply"><label>Reply<small>as '+esc(o.who)+'</small></label><textarea class="in" rows="3" id="thRep" placeholder="Type a reply"></textarea>'
      +'<div class="act" style="display:flex; gap:14px; align-items:center; margin-top:12px"><button class="btn pri" id="thSend">Send</button>'
      +(cur.st==='done'?'<span class="t13" style="color:var(--g550)">Resolved · reopen by replying</span>':'<span class="lnk" id="thRes">Mark resolved</span>')+'</div></div></div>';
   } else thread='<div class="blank" style="margin-top:0"><div class="t">Nothing here.</div><p>'+esc(o.empty||'Every thread is either with the other side or closed.')+'</p></div>';
   o.host.innerHTML='<div class="mp"><div class="list">'+(list||'<div class="blank" style="margin-top:0; padding:30px 16px"><div class="t">'+esc(o.emptyT||'Nothing here.')+'</div><p>'+esc(o.empty||'')+'</p></div>')+'</div><div class="pane">'+thread+'</div></div>';
   $$('.mp .list a').forEach(function(a){ a.addEventListener('click',function(){ o.onOpen(a.dataset.t); o.redraw(); }) });
   if($('#thSend')) $('#thSend').addEventListener('click',function(){ var v=$('#thRep').value.trim(); if(!v){ toast('Nothing to send yet.'); return; } o.reply(cur,v); });
   if($('#thRes')) $('#thRes').addEventListener('click',function(){ o.resolve(cur); });
 }
