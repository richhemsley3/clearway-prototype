 function guideBody(g){
   return '<div class="gd">'+g.body.map(function(b){
     if(b[0]==='h2') return '<h2>'+esc(b[1])+'</h2>';
     if(b[0]==='p') return '<p>'+b[1]+'</p>';
     if(b[0]==='ul') return '<ul>'+b[1].map(function(x){return '<li>'+x+'</li>'}).join('')+'</ul>';
     if(b[0]==='ol') return '<ol>'+b[1].map(function(x){return '<li>'+x+'</li>'}).join('')+'</ol>';
     if(b[0]==='code') return '<div class="code">'+esc(b[1].replace(/https:\/\/api\.gpn\.dev/g,'https://a41f-sandbox.gpn-api.net').replace(/sk_sandbox_4f2c…c91e/g,'gpn_sand_a41f…9d2c'))+'</div>';
     if(b[0]==='note') return '<div class="note plain"><p>'+b[1]+'</p></div>';
     return '';
   }).join('')+'</div>';
 }
 function openGuide(k){
   var g=GUIDES[k]; if(!g) return;
   var i=GUIDE_ORDER.indexOf(k), nx=GUIDE_ORDER[(i+1)%GUIDE_ORDER.length];
   panel(g.t, g.crumb.join(' · ')+' · '+(i+1)+' of '+GUIDE_ORDER.length,
     '<p style="font-size:16px; color:var(--g650); line-height:1.5">'+esc(g.deck)+'</p>'+guideBody(g),
     'Next: '+GUIDES[nx].t, function(){ openGuide(nx) }, 'go');
 }
