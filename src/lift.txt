 /* ── the instrument ───────────────────────────────────────────── */
 /* o: {fig, unit, cap, segs:[[label, value, tone]], legend:true} · tones: navy | warn | bad | good | rest */
 function instrument(o){
   var tot=o.segs.reduce(function(a,s){return a+s[1]},0)||1;
   var TONE={navy:'var(--navy)', warn:'var(--warn)', bad:'var(--bad)', good:'var(--good)', rest:'var(--g200)'};
   return '<div class="inst2">'
    +'<div class="fig">'+esc(String(o.fig))+(o.unit?'<small>'+esc(o.unit)+'</small>':'')+'</div>'
    +(o.cap?'<div class="cap">'+esc(o.cap)+'</div>':'')
    +'<div class="bar">'+o.segs.map(function(s){ return s[1]?'<i style="flex:'+s[1]+' 1 0; background:'+TONE[s[2]||'rest']+'" title="'+esc(s[0])+' · '+s[1]+'"></i>':'' }).join('')+'</div>'
    +'<div class="leg">'+o.segs.filter(function(s){return s[1]>0}).map(function(s){ return '<span><i style="background:'+TONE[s[2]||'rest']+'"></i>'+esc(s[0])+' <b>'+s[1]+'</b></span>' }).join('')+'</div>'
    +'</div>';
 }
 /* put an instrument at the head of a brief's right column, keeping whatever record was there */
 function liftBrief(sel, html, replace, dropAlso){
   var now=$(sel); if(!now) return;
   var right=now.children[1]; if(!right) return;
   if(dropAlso){ var al=now.querySelector('.also'); if(al) al.remove(); }
   if(replace || right.classList.contains('inst')) right.outerHTML='<div class="col">'+html+'</div>';
   else right.outerHTML='<div class="col">'+html+right.outerHTML+'</div>';
 }
 /* a working set gets a ground; the element keeps its place */
 function panelize(el, cls){
   if(!el || el.closest('.surface')) return;
   var p=document.createElement('div'); p.className='surface set'+(cls?' '+cls:''); el.parentNode.insertBefore(p, el); p.appendChild(el);
   el.style.marginTop='0';
 }
