   if(risk.dstate!=='done') m.push({t:'Chase second-line risk sign-off', s:'Ridgeline Freight · ours · day 6 against a 4-day median', kind:'Chase', due:'Today', d:'wa', act:'Chase risk',

     why:'The passport expires on 12 September. Without a replacement on file, settlement pauses on the first cycle after that date — which would be Northgate’s first cycle.',
     go:function(){ openRecord(0) }, run:function(){ toast('Chase sent to Halvard Ruiz, copied to Idris Vance.'); }});

   if(f) m.push({t:f+(f===1?' certification case is':' certification cases are')+' failing', s:'Northgate Financial · the window opens 15 September · AUTH-14 decides it', kind:'Watch', due:'15 Sep', d:'nt', go:function(){ inTab='cert'; go('pd-integr') }});
   m.push({t:'Tell Idris the mutual TLS certificate expires 21 September', s:'Northgate Financial · fifteen days before go live · nobody there has raised it', kind:'Tell', due:'This week', d:'nt', go:function(){ go('pd-integr') }});

 /* ── the agent's messages: every thread, from our side ──────────── */
 var agentMsgOpen=null, agentMsgF='all';
 function drawAgentMsgs(){
   var all=THREAD.slice();
   var awaiting=function(t){ return t.st!=='done' && t.msgs[0][1]!=='Global Payment Network' };
   var yours=all.filter(awaiting);
   $('#amSub').textContent=all.length+' threads across three accounts · '+yours.length+' awaiting your reply · a thread is attached to the record it is about';
   var K=[['all','All threads',all.length],['you','Awaiting your reply',yours.length],['theirs','With them',all.filter(function(t){return !awaiting(t)&&t.st!=='done'}).length],['done','Resolved',all.filter(function(t){return t.st==='done'}).length]];
   $('#amSeg').innerHTML='<span class="seg">'+K.map(function(t){ return '<button data-m="'+t[0]+'" aria-pressed="'+(agentMsgF===t[0])+'">'+t[1]+'<span class="ct">'+t[2]+'</span></button>' }).join('')+'</span>';
   $$('#amSeg button').forEach(function(b){ b.addEventListener('click',function(){ anchored($('#amSeg'), function(){ agentMsgF=b.dataset.m; agentMsgOpen=null; drawAgentMsgs(); }) }) });
   var rows=all.filter(function(t){ return agentMsgF==='all' || (agentMsgF==='you'&&awaiting(t)) || (agentMsgF==='theirs'&&!awaiting(t)&&t.st!=='done') || (agentMsgF==='done'&&t.st==='done') });
   if(!rows.length){ $('#amBody').innerHTML='<div class="blank" style="margin-top:22px"><div class="t">Nothing here.</div><p>Every thread is with the participant or closed.</p></div>'; return; }
   var h='<table class="tb" style="margin-top:22px"><thead><tr><th>Thread</th><th class="opt">Account</th><th class="opt2">About</th><th>Status</th><th class="n">Last</th></tr></thead><tbody>';
   rows.forEach(function(t){
     var on=agentMsgOpen===t.id, aw=awaiting(t);
     h+='<tr data-t="'+t.id+'"'+(on?' class="sel"':'')+'><td>'+esc(t.t)+'<span class="s2">'+t.msgs.length+(t.msgs.length===1?' message':' messages')+' · '+esc(t.w)+'</span></td>'
      +'<td class="opt mute">'+esc(ACC[t.a].n)+'</td><td class="opt2 mute nw">'+esc(t.about||t.id)+'</td>'
      +'<td><span class="st '+(t.st==='done'?'ok':aw?'wa':'nt')+'">'+(t.st==='done'?'Resolved':aw?'Awaiting your reply':'With '+ACC[t.a].n.split(' ')[0])+'</span></td><td class="n mute nw">'+esc(t.last)+'</td></tr>';
     if(on){
       h+='<tr class="expand"><td colspan="5"><div style="max-width:74ch">'
        +t.msgs.map(function(m){ var us=m[1]==='Global Payment Network';
          return '<div style="padding:14px 0; border-bottom:1px solid var(--g100)"><div class="t13" style="color:var(--g550)">'+esc(m[0])+' · '+esc(m[1])+' · '+esc(m[2])+'</div><p style="font-size:15px; line-height:1.55; margin-top:6px'+(us?'':'; color:var(--g650)')+'">'+esc(m[3])+'</p></div>' }).join('')
        +'<div class="field" style="margin-top:18px"><label>Reply<small>as Dara Whitfield</small></label><textarea class="in" rows="3" id="amRep" placeholder="Type a reply"></textarea></div>'
        +'<div class="act" style="display:flex; gap:14px; align-items:center; margin-top:14px"><button class="btn pri" id="amSend">Send</button>'+(t.st==='done'?'':'<span class="lnk" id="amRes">Mark resolved</span>')+'</div></div></td></tr>';
     }
   });
   $('#amBody').innerHTML=h+'</tbody></table>';
   $$('#amBody tr[data-t]').forEach(function(tr){ tr.addEventListener('click',function(){ anchored($('#amBody'), function(){ agentMsgOpen=agentMsgOpen===tr.dataset.t?null:tr.dataset.t; drawAgentMsgs(); }) }) });
   if($('#amSend')) $('#amSend').addEventListener('click',function(e){ e.stopPropagation();
     var t=all.filter(function(x){return x.id===agentMsgOpen})[0], v=$('#amRep').value.trim(); if(!v){ toast('Nothing to send yet.'); return; }
     t.msgs.unshift(['Dara Whitfield','Global Payment Network','Just now',v]); t.st='you'; t.last='Just now'; drawAgentMsgs(); refreshNav();
     toast('Sent to '+ACC[t.a].who+'. It is on their screen against '+(t.about||t.id).split(' · ')[0].toLowerCase()+'.', function(){ t.msgs.shift(); drawAgentMsgs(); refreshNav(); }); });
   if($('#amRes')) $('#amRes').addEventListener('click',function(e){ e.stopPropagation();
     var t=all.filter(function(x){return x.id===agentMsgOpen})[0], was=t.st; t.st='done'; drawAgentMsgs(); refreshNav();
     toast('Resolved. Either side can reopen it for seven days.', function(){ t.st=was; drawAgentMsgs(); refreshNav(); }); });
 }
